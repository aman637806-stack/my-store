<script>
let selectedVariant = 'Flame Orange';

function switchImage(btn, src) {
document.getElementById('mainImage').src = src;
document.querySelectorAll('button img').forEach(img => {
img.parentElement.classList.remove('border-brand-orange', 'ring-2', 'ring-brand-orange/50');
img.parentElement.classList.add('border-transparent');
});
btn.classList.remove('border-transparent');
btn.classList.add('border-brand-orange', 'ring-2', 'ring-brand-orange/50');
}

function selectVariant(variant, btn) {
selectedVariant = variant;
document.getElementById('selectedVariantText').innerText = 'Style: ' + variant;
document.querySelectorAll('.variant-btn').forEach(b => {
b.classList.remove('border-brand-orange', 'bg-brand-orange/10');
b.classList.add('border-gray-700', 'bg-gray-800/50');
});
btn.classList.remove('border-gray-700', 'bg-gray-800/50');
btn.classList.add('border-brand-orange', 'bg-brand-orange/10');
}

function openCheckout() {
document.getElementById('checkoutModal').classList.remove('hidden');
}

function closeCheckout() {
document.getElementById('checkoutModal').classList.add('hidden');
}

function validateAndConfirmOrder(event) {
event.preventDefault();

const name = document.getElementById('fullName').value.trim();
const phone = document.getElementById('phoneNumber').value.trim();
const pincode = document.getElementById('pincode').value.trim();
const address = document.getElementById('address').value.trim();
const utr = document.getElementById('utrNumber').value.trim();
const statusMsg = document.getElementById('formStatusMsg');
const btn = document.getElementById('confirmOrderBtn');

// 1. Validate Delivery Details
if (!name || !phone || !pincode || !address) {
statusMsg.innerHTML = `<span class="text-red-400 font-bold">❌ Kripya apni saari delivery details (Name, Phone, Pincode, Address) bhariye!</span>`;
return;
}

// 2. Validate UTR Length
if (!utr || utr.length < 8) {
statusMsg.innerHTML = `<span class="text-red-400 font-bold">❌ Kripya valid UPI Transaction Ref / UTR Number enter karein!</span>`;
document.getElementById('utrNumber').focus();
return;
}

// 3. Process Verification Loader
btn.disabled = true;
btn.innerHTML = `<i class="fa-solid fa-spinner animate-spin"></i><span>Verifying Address & Bank UTR...</span>`;
statusMsg.innerHTML = `<span class="text-yellow-400 font-semibold animate-pulse">⏳ Checking delivery location & UTR payment confirmation...</span>`;

// 4. Simulated Success Handshake
setTimeout(() => {
closeCheckout();
document.getElementById('lblOrderId').innerText = Math.floor(10000 + Math.random() * 90000);
document.getElementById('lblCustomerName').innerText = name;
document.getElementById('lblCustomerPhone').innerText = phone;
document.getElementById('lblOrderVariant').innerText = selectedVariant;
document.getElementById('lblUtrCode').innerText = utr;
document.getElementById('successModal').classList.remove('hidden');
}, 3000);
}
</script>
